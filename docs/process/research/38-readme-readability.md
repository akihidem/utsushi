## 文の長さと読点を数えた道具

```python
"""yomiyasu の原則 7（平均の文の長さ 30〜45 字・1 文の読点 0〜2 個）を、README の節ごとに測る。lint はこの 2 つを数えないので補う。
コードの囲み・表・HTML の印・リンクの先は数えない。"""
import re, sys
text = open(sys.argv[1], encoding="utf-8").read()
text = re.sub(r"```.*?```", "", text, flags=re.S)
text = re.sub(r"<!--.*?-->", "", text, flags=re.S)
sec, out = "(頭)", {}
for ln in text.splitlines():
    if ln.startswith("#"):
        sec = ln.lstrip("# ").strip()[:24]
        continue
    if ln.startswith("|") or not ln.strip() or ln.strip().startswith("<"):
        continue
    ln = re.sub(r"\]\([^)]*\)", "]", ln)          # リンクの先
    ln = re.sub(r"`[^`]*`", "X", ln)               # コードは 1 字
    ln = re.sub(r"^\s*[-*>] ", "", ln)
    ln = re.sub(r"^\*\*[^*]{1,16}\*\*[:：] ?", "", ln).replace("**", "")  # 箇条の頭の見出し語とコロンは文の字に数えない（yomiyasu も見出し語として残す）
    for s in re.split(r"(?<=。)", ln):
        s = s.strip()
        if len(s) >= 4:
            out.setdefault(sec, []).append(s)
tot = []
print(f"{'節':<26}{'文':>4}{'平均字':>7}{'読点3+':>7}{'60字+':>6}")
for k, ss in out.items():
    n = [len(s) for s in ss]; c = [s.count("、") for s in ss]
    tot += ss
    print(f"{k:<26}{len(ss):>4}{sum(n)/len(n):>7.1f}{sum(x >= 3 for x in c):>7}{sum(x > 60 for x in n):>6}")
n = [len(s) for s in tot]
print(f"{'全体':<26}{len(tot):>4}{sum(n)/len(n):>7.1f}{sum(s.count('、') >= 3 for s in tot):>7}{sum(x > 60 for x in n):>6}")
```
